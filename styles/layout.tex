% ------------------------------------------------------------
%  Numeración de páginas
%  IEEEtran en modo "conference" vacía cabeceras y pies, así que
%  el documento salía SIN números de página. Se define un pie:
%     izquierda: sección actual      derecha: Pág. N
%  La portada (titlepage) queda sin número y el índice es la pág. 1,
%  que es exactamente lo que referencia \tableofcontents.
% ------------------------------------------------------------
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0.4pt}
\renewcommand{\sectionmark}[1]{\markboth{\thesection\ #1}{}}
\fancyfoot[L]{\footnotesize\nouppercase{\leftmark}}
\fancyfoot[R]{\footnotesize P\'ag.\ \thepage}

% ------------------------------------------------------------
%  Índice compacto
%  IEEEtran deja 1em en blanco antes de cada capítulo del índice; con 10
%  capítulos eso lo desborda a una segunda página. Misma definición que
%  IEEEtran.cls, cambiando solo ese espacio a 0.4em.
% ------------------------------------------------------------
\makeatletter
\def\l@section#1#2{\addpenalty{\@secpenalty}\addvspace{0.4em plus 1pt}%
    \@tempdima 2.75em \begingroup \parindent \z@ \rightskip \@pnumwidth%
    \parfillskip-\@pnumwidth {\bfseries\leavevmode #1}\hfil\hbox to\@pnumwidth{\hss #2}\par%
    \endgroup}
\makeatother

% ------------------------------------------------------------
%  Ecuaciones en display: menos aire arriba/abajo.
%  IEEEtran usa 1.5ex (~7pt) a cada lado; en dos columnas es mucho.
% ------------------------------------------------------------
\setlength{\abovedisplayskip}{4pt plus 2pt minus 2pt}
\setlength{\belowdisplayskip}{4pt plus 2pt minus 2pt}
\setlength{\abovedisplayshortskip}{2pt plus 1pt}
\setlength{\belowdisplayshortskip}{2pt plus 1pt}
